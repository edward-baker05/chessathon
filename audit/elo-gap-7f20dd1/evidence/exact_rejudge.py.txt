from pathlib import Path
import sys,json
ROOT=Path(__file__).resolve().parents[1];sys.path.insert(0,str(ROOT))
import chess,chess.engine
from uci_exact import Engine
OUT=Path(__file__).resolve().parent
cases=json.loads((OUT/'current-failures.json').read_text());abl=json.loads((OUT/'selective-ablation.json').read_text());out=[]
sf=Engine('/home/edwardb/Documents/stockfish/stockfish-linux-x86-64-universal');sf.configure({'Threads':1,'Hash':32})
for idx,p in enumerate(cases):
 b=chess.Board(p['source']['fen']);choices=set(p['judged'])
 for a in abl:
  if a['case_index']==idx:choices.update(a['judged'])
 row={'index':idx,'fen':b.fen(),'moves':{}}
 for uci in sorted(choices):
  sf.configure({'Clear Hash':None});i=sf.analyse(b,chess.engine.Limit(nodes=1000000),root_moves=[chess.Move.from_uci(uci)])
  s=i['score'].pov(b.turn);last=sf.last_reported
  row['moves'][uci]={'cp':s.score(),'mate':s.mate(),'depth':i['depth'],'nodes_at_exact':i.get('nodes'),'last_was_bound':bool(last.get('upperbound') or last.get('lowerbound')),'skipped_bounds':sf.skipped_bounds,'pv':[m.uci() for m in i['pv']]}
 out.append(row);print(idx,{m:v['cp'] for m,v in row['moves'].items()},flush=True);(OUT/'exact-reference.json').write_text(json.dumps(out,indent=2))
strong=[]
for idx,moves in [(0,['h8g7','g5h4']),(1,['c2e2','e1c1']),(4,['f1g3','e3f2']),(11,['e5e4','h5h4','e8e6','f5e3'])]:
 b=chess.Board(cases[idx]['source']['fen']);row={'index':idx,'fen':b.fen(),'moves':{}}
 for uci in moves:
  sf.configure({'Clear Hash':None});i=sf.analyse(b,chess.engine.Limit(nodes=4000000),root_moves=[chess.Move.from_uci(uci)])
  s=i['score'].pov(b.turn);last=sf.last_reported
  row['moves'][uci]={'cp':s.score(),'mate':s.mate(),'depth':i['depth'],'nodes_at_exact':i.get('nodes'),'last_was_bound':bool(last.get('upperbound') or last.get('lowerbound')),'pv':[m.uci() for m in i['pv']]}
 strong.append(row);print('strong',idx,{m:v['cp'] for m,v in row['moves'].items()},flush=True);(OUT/'exact-reference-4m.json').write_text(json.dumps(strong,indent=2))
sf.quit()
